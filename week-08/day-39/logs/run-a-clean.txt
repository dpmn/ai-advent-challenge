func (s *Server) handleFeedback(w http.ResponseWriter, r *http.Request) {
    var req feedbackRequest
    if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
        writeError(w, http.StatusBadRequest, "некорректный JSON запрос")
        return
    }

    if req.Rating < 1 || req.Rating > 5 {
        writeError(w, http.StatusBadRequest, "rating должен быть целым числом от 1 до 5")
        return
    }

    // Get path from env
    feedbackPath := os.Getenv("FEEDBACK_PATH")
    if feedbackPath == "" {
        feedbackPath = "./feedback.jsonl"
    }

    // Create timestamp
    timestamp := time.Now().Format(time.RFC3339)

    // Write to file
    file, err := os.OpenFile(feedbackPath, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0644)
    if err != nil {
        writeError(w, http.StatusInternalServerError, "ошибка открытия файла")
        return

    defer file.Close()

    // Write entry
    entry := fmt.Sprintf(`{"recipe_id": "%s", "rating": %d, "comment": "%s", "timestamp": "%s"}`,

    _, err = file.WriteString(entry + "\n")

    if err != nil {
        writeError(w, http.StatusInternalServerError, "ошибка записи в файл")
        return
    }

    writeJSON(w, http.StatusCreated, map[string]string{"status": "ok"})
}
