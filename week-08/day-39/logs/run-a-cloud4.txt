+type feedbackRequest struct {
+	RecipeID *string `json:"recipe_id"`
+	Rating   int     `json:"rating"`
+	Comment  *string `json:"comment"`
+}
+
+type feedbackResponse struct {
+	Status string `json:"status"`

+// handleFeedback сохраняет обратную связь в JSONL-файл.
+func (s *Server) handleFeedback(w http.ResponseWriter, r *http.Request) {
+	var req feedbackRequest
+	if !decode(w, r, &req) {
+		return
+	}
+	if req.Rating < 1 || req.Rating > 5 {
+		writeError(w, http.StatusBadRequest, "rating должен быть от 1 до 5")
+		return
+	}
+	if req.Comment != nil && len(*req.Comment) > 1000 {
+		writeError(w, http.StatusBadRequest, "comment не длиннее 1000 символов")
+		return
+	}
+
+	path := os.Getenv("FEEDBACK_PATH")
+	if path == "" {
+		path = "./feedback.jsonl"
+	}
+
+	record := map[string]any{
+		"recipe_id":  req.RecipeID,
+		"rating":     req.Rating,
+		"comment":    req.Comment,
+		"timestamp":  time.Now().UTC().Format(time.RFC3339),
+	}
+	data, err := json.Marshal(record)
+	if err != nil {
+		log.Printf("feedback: маршалинг: %v", err)
+		writeError(w, http.StatusInternalServerError, "внутренняя ошибка")
+		return
+	}
+	data = append(data, '\n')
+
+	f, err := os.OpenFile(path, os.O_CREATE|os.O_WRONLY|os.O_APPEND, 0o644)
+	if err != nil {
+		log.Printf("feedback: открытие %s: %v", path, err)
+		writeError(w, http.StatusInternalServerError, "внутренняя ошибка")
+		return
+	}
+	_, err = f.Write(data)
+	closeErr := f.Close()
+	if err != nil {
+		log.Printf("feedback: запись: %v", err)
+		writeError(w, http.StatusInternalServerError, "внутренняя ошибка")
+		return
+	}
+	if closeErr != nil {
+		log.Printf("feedback: закрытие: %v", closeErr)
+		writeError(w, http.StatusInternalServerError, "внутренняя ошибка")
+		return
+	}
+
+	writeJSON(w, http.StatusCreated, feedbackResponse{Status: "ok"})
+}
