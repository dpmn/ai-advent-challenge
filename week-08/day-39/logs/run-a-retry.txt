package api

import (
    "encoding/json"
    "os"
    "time"
)

// feedbackRecord represents a feedback record.
type feedbackRecord struct {
    Rating   int    `json:"rating"`
    Comment  string `json:"comment,omitempty"`
    RecipeID string `json:"recipe_id,omitempty"`
    Timestamp string `json:"timestamp"`
}

// handleFeedback handles POST /api/v1/feedback.
func (s *Server) handleFeedback(w http.ResponseWriter, r *http.Request) {
    // Decode the request body
    var req feedbackRecord
    err := json.NewDecoder(r.Body).Decode(&req)
    if err != nil {
        w.WriteHeader(http.StatusBadRequest)
        json.NewEncoder(w).Encode(map[string]string{"error": err.Error()})
        return
    }

    // Validate rating
    if req.Rating < 1 || req.Rating > 5 {
        w.WriteHeader(http.StatusBadRequest)
        json.NewEncoder(w).Encode(map[string]string{"error": "rating must be between 1 and 5"})
        return
    }

    // Create feedback record with timestamp
    record := feedbackRecord{
        Rating:   req.Rating,
        Comment:  req.Comment,
        RecipeID: req.RecipeID,
        Timestamp: time.Now().Format(time.RFC3339),
    }

    // Write to feedback file
    feedbackPath := os.Getenv("FEEDBACK_PATH")
    if feedbackPath == "" {
        feedbackPath = "./feedback.jsonl"
    }

    file, err := os.OpenFile(feedbackPath, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0644)
    if err != nil {
        w.WriteHeader(http.StatusInternalServerError)
        json.NewEncoder(w).Encode(map[string]string{"error": "failed to open feedback file: " + err.Error()})
        return
    }
    defer file.Close()

    // Marshal and write the record
    if err := json.NewEncoder(file).Encode(record); err != nil {
        w.WriteHeader(http.StatusInternalServerError)
        json.NewEncoder(w).Encode(map[string]string{"error": "failed to write feedback record: " + err.Error()})
        return
    }

    w.WriteHeader(http.StatusCreated)
    json.NewEncoder(w).Encode(map[string]string{"status": "ok"})
}
